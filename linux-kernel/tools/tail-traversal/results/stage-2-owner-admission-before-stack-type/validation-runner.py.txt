paths += ['src/topal-semantics/se-requirements.md', 'src/topal-semantics/src/lib.rs', 'src/topal-semantics/Cargo.toml', 'src/topal-semantics/src/systems_storage.rs', 'src/topal-semantics/src/systems_storage_tail.rs', 'src/topal-semantics/src/systems_region_library.rs', 'src/topal-semantics/src/systems_region_owner.rs', 'src/topal-semantics/src/systems_tail_session.rs', 'src/topal-semantics/src/systems_tail_values.rs', 'src/topal-semantics/src/systems_tail_session_tests.rs', 'src/topal-compiler/src/systems_tail_model.rs', 'linux-kernel/design/tail-session-model.md']

record = {
    'schema': 'topal.tail-owner-admission-validation/1', 'complete': False, 'passed': False,
    'base_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip(),

snapshot_dir = out / 'source-snapshots'
snapshot_dir.mkdir()
def save_bytes(directory, name, raw):
    compressed = gzip.compress(raw, mtime=0)
    destination = directory / (name + '.gz')
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(compressed)
    assert gzip.decompress(compressed) == raw
    return {'file': str(destination.relative_to(out)), 'sha256': sha(compressed), 'uncompressed_sha256': sha(raw)}
def snapshot(path):
    raw = (root / path).read_bytes()
    return save_bytes(snapshot_dir, sha(raw) + '.txt', raw)
record['source_snapshots'] = {path: snapshot(path) for path in paths}
record['verification_inputs'] = {path: snapshot(path) for path in ['se/traceability.md', 'se/tool-conformance.md', 'se/test-strategy.md']}
for tool in ['rustc', 'cargo', 'cc']:
    record[tool] = subprocess.check_output([tool, '--version'], cwd=root, text=True).splitlines()[0]
group = next(line.split(':', 2)[2] for line in Path('/proc/self/cgroup').read_text().splitlines() if line.startswith('0::'))
cgroup = Path('/sys/fs/cgroup') / group.lstrip('/')
record['cgroup'] = {'path': group, 'memory.max': (cgroup / 'memory.max').read_text().strip(),
                   'memory.swap.max': (cgroup / 'memory.swap.max').read_text().strip()}
assert record['cgroup']['memory.swap.max'] == '0'

    ('source-size', ['python3', 'scripts/check_rust_file_sizes.py']),
]
seen = set()
def collect_native(command_name):
    for metadata_path in sorted(scratch.glob('tail-step-*/component.json')):
        if str(metadata_path) in seen:
            continue
        seen.add(str(metadata_path))
        directory = metadata_path.parent
        metadata = json.loads(metadata_path.read_text())
        assert metadata['llvm_sha256'] == sha((directory / 'step.ll').read_bytes())
        item = {'command': command_name, 'cache_directory': str(directory), 'metadata': metadata,
                'snapshots': {}, 'binary_identities': {}}
        for filename in ['step.ll', 'tail_step_bindings.h', 'component.json', 'runtime.ll', 'runtime.json', 'native-inspection.json'] + [label + suffix for label in ['llc', 'runtime-llc', 'source-disassembly', 'runtime-disassembly', 'cc', 'native', 'gdb'] for suffix in ['.stdout.log', '.stderr.log', '.command.json']]:
            path = directory / filename
            if path.exists():
                item['snapshots'][filename] = save_bytes(out / 'native-components' / directory.name, filename, path.read_bytes())
        for filename in ['step.o', 'client', 'boundary.o', 'runtime.o']:
            path = directory / filename
            if path.exists():
                raw = path.read_bytes()
                item['binary_identities'][filename] = {'sha256': sha(raw), 'bytes': len(raw), 'retention': 'external disk cache; exact LLVM/C/header/command inputs archived'}
        record['native_components'].append(item)

owner_seen = set()
def collect_owner(command_name):
    for metadata_path in sorted(scratch.glob("tail-owner-*/owner.json")):
        if str(metadata_path) in owner_seen: continue
        owner_seen.add(str(metadata_path))
        directory = metadata_path.parent
        metadata = json.loads(metadata_path.read_text())
        item = {"command":command_name,"cache_directory":str(directory),"metadata":metadata,"snapshots":{},"binary_identities":{}}
        for path in sorted(directory.iterdir()):
            if path.name in ["source.o","runtime.o","owner.o","boundary.o","client"]:
                raw = path.read_bytes()
                item["binary_identities"][path.name] = {"sha256":sha(raw),"bytes":len(raw),"retention":"external disk cache; exact compiler/C/source/runtime/controller/command inputs archived"}
            else:
                item["snapshots"][path.name] = save_bytes(out / "owner-components" / directory.name,path.name,path.read_bytes())
        record["owner_components"].append(item)

def save():
    record['cgroup']['memory.peak'] = (cgroup / 'memory.peak').read_text().strip()
    record['cgroup']['memory.events'] = (cgroup / 'memory.events').read_text()
    target.write_text(json.dumps(record, indent=2) + '\n')
save()
print('TAIL_COMPONENTS_STARTED ' + run_id, flush=True)
for name, argv in commands:
    print('RUNNING ' + name, flush=True)
    started = time.monotonic()
    plain = scratch / (name + '.log')
    with plain.open('wb') as log:
        result = subprocess.run(argv, cwd=root, env=env, stdout=log, stderr=subprocess.STDOUT)
    log_record = save_bytes(out, name + '.log', plain.read_bytes())
    record['commands'].append({'name': name, 'argv': argv, 'exit_code': result.returncode,
        'elapsed_seconds': time.monotonic() - started, 'log': log_record['file'],
        'encoding': 'gzip', 'sha256': log_record['sha256'], 'uncompressed_sha256': log_record['uncompressed_sha256']})
    collect_native(name)
    collect_owner(name)
    save()
    print(name + ' EXIT ' + str(result.returncode), flush=True)
    if result.returncode:
        record['complete'] = True
        save()
        raise SystemExit(result.returncode)
assert record['sources'] == {path: sha((root / path).read_bytes()) for path in paths}, 'frozen source changed'
for path, item in record['verification_inputs'].items():
    assert sha((root / path).read_bytes()) == item['uncompressed_sha256'], 'verification input changed'
record.update(complete=True, passed=True)
save()
print('TAIL_COMPONENTS_VALIDATION_PASS', flush=True)
