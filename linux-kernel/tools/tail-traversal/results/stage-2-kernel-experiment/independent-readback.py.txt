from pathlib import Path
import gzip
import hashlib
import json
import re

root = Path('/home/mats/git/topal')
out = root / 'linux-kernel/tools/tail-traversal/results/stage-2-kernel-experiment'
record = json.loads((out / 'qualification.json').read_text())
sha = lambda data: hashlib.sha256(data).hexdigest()
assert record['complete'] and record['passed']
assert record['cgroup']['memory.swap.max'] == '0'
def read(item):
    compressed = (out / item['file']).read_bytes()
    assert sha(compressed) == item['sha256']
    raw = gzip.decompress(compressed)
    assert sha(raw) == item['uncompressed_sha256']
    return raw
for path, item in record['source_snapshots'].items():
    raw = read(item)
    assert sha(raw) == record['sources'][path] == sha((root / path).read_bytes())
for item in record['verification_inputs'].values():
    read(item)
assert sha((out / 'validation-runner.py.txt').read_bytes()) == record['runner_sha256']
for command in record['commands']:
    assert command['exit_code'] == 0
    raw = read(dict(file=command['log'], sha256=command['sha256'], uncompressed_sha256=command['uncompressed_sha256']))
    summaries = [line for line in raw.decode().splitlines() if line.startswith('test result:')]
    print(command['name'], command['exit_code'], summaries)
totals = {}
warnings = []
runtime_facts = {}
for component in record['native_components']:
    snapshots = {name: read(item) for name, item in component['snapshots'].items()}
    assert sha(snapshots['step.ll']) == component['metadata']['llvm_sha256']
    assert json.loads(snapshots['component.json']) == component['metadata']
    for name in [label for label in ['llc', 'runtime-llc', 'source-disassembly', 'runtime-disassembly', 'cc', 'native', 'gdb'] if label + '.command.json' in snapshots]:
        assert json.loads(snapshots[name + '.command.json'])['exit_code'] == 0
        if snapshots[name + '.stderr.log']:
            warnings.append((component['cache_directory'], name, snapshots[name + '.stderr.log'].decode()))
    for name, identity in component['binary_identities'].items():
        raw = (Path(component['cache_directory']) / name).read_bytes()
        assert sha(raw) == identity['sha256'] and len(raw) == identity['bytes']
    line = snapshots['native.stdout.log'].decode().strip()
    assert line.startswith('TAIL_STEP_NATIVE_PASS ')
    facts = {key: int(value) for key, value in re.findall(r'(\w+)=(\d+)', line)}
    total = totals.setdefault(component['command'], {'components': 0, 'cases': 0, 'calls': 0, 'checkpoints': 0, 'fatal': 0})
    total['components'] += 1
    for key, value in facts.items():
        total[key] = total.get(key, 0) + value
    if 'gdb.stdout.log' in snapshots:
        assert b'TAIL_BOUNDARY_GDB_PASS' in snapshots['gdb.stdout.log']
        total['gdb'] = total.get('gdb', 0) + 1
    if component['metadata'].get('native_provider_hooks'):
        runtime = json.loads(snapshots['runtime.json'])
        assert sha(snapshots['runtime.ll']) == runtime['llvm_sha256']
        assert runtime['boundary_sha256'] == component['binary_identities']['boundary.o']['sha256']
        assert runtime['source_step']['llvm_sha256'] == component['metadata']['llvm_sha256']
        layout = runtime['layout']
        assert layout['minimum_temporary_bytes'] == layout['context_bytes'] + 72 <= 65536
        assert layout['history_bytes'] == 40 and layout['frame_bytes'] == 88
        assert layout['caller_output_frame_bytes'] == 56
        native = json.loads(snapshots['native-inspection.json'])
        assert native['maximum_source_stack_bytes'] <= 4096
        assert native['maximum_checkpoint_stack_bytes'] <= 4096
        assert native['maximum_completion_stack_bytes'] <= 4096
        assert set(native['runtime_imports']) == {'topal_tail_step_component_v1', 'topal_tail_readonly_bytes_v1',
            'topal_x86_tail_boundary_active_v1', 'topal_x86_tail_boundary_abort_v1'}
        paths = native['reached_call_paths']
        assert paths['topal_x86_tail_boundary_abort_v1'] == 8
        for function, bound in paths.items():
            if function == 'topal_x86_tail_boundary_abort_v1': continue
            expected = native['frames'][function] + 8 + max((paths[callee] for callee in native['calls'][function]), default=0)
            assert bound == expected
        assert paths['topal_tail_step_component_v1'] == native['maximum_source_stack_bytes']
        assert paths['topal_tail_checkpoint_body_v1'] == native['maximum_checkpoint_stack_bytes']
        assert paths['topal_tail_complete_body_v1'] == native['maximum_completion_stack_bytes']
        group = runtime_facts.setdefault(component['command'], {'clients':0,'stack':set(),'checkpoint_stack':set(),'completion_stack':set(),'context':set(),'faults':{},'boundary_faults':{},'completion_faults':{},'gdb':0})
        group['clients'] += 1
        group['stack'].add(native['maximum_source_stack_bytes'])
        group['checkpoint_stack'].add(native['maximum_checkpoint_stack_bytes'])
        group['completion_stack'].add(native['maximum_completion_stack_bytes'])
        group['context'].add(layout['context_bytes'])
        fault = component['metadata'].get('native_source_fault_fixture', 0)
        group['faults'][fault] = group['faults'].get(fault,0) + 1
        boundary_fault = component['metadata'].get('native_boundary_fault_fixture', 0)
        group['boundary_faults'][boundary_fault] = group['boundary_faults'].get(boundary_fault,0) + 1
        completion_fault = component['metadata'].get('native_completion_fault_fixture', 0)
        group['completion_faults'][completion_fault] = group['completion_faults'].get(completion_fault,0) + 1
        group['gdb'] += int('gdb.stdout.log' in snapshots)
print('NATIVE_TOTALS', totals)
print('PROVIDER_FACTS', runtime_facts)
print('SOURCES', len(record['sources']), 'VERIFICATION', len(record['verification_inputs']), 'LOGS', len(record['commands']))
print('CGROUP', record['cgroup'])
assert all(name == 'gdb' and message == '\nwarning: .debug_names not created by gdb; ignoring\n' for _, name, message in warnings)
print('TOOL_WARNINGS', len(warnings), 'known GDB .debug_names compatibility warnings only')

owner_totals = {}
owner_warnings = []
for item in record['owner_components']:
    snapshots = {name: read(entry) for name, entry in item['snapshots'].items()}
    metadata = item['metadata']
    assert json.loads(snapshots['owner.json']) == metadata
    assert sha(snapshots['owner.ll']) == metadata['llvm_sha256']
    runtime = metadata['runtime']
    assert sha(snapshots['runtime.ll']) == runtime['llvm_sha256']
    assert sha(snapshots['source.ll']) == runtime['source_step']['llvm_sha256']
    assert runtime['boundary_sha256'] == item['binary_identities']['boundary.o']['sha256']
    assert metadata['caller_input_frame_bytes'] == 24
    assert runtime['layout']['caller_output_frame_bytes'] == 56
    for name, raw in snapshots.items():
        if name.endswith('.command.json'): assert json.loads(raw)['exit_code'] == 0
        if name.endswith('.stderr.log') and raw: owner_warnings.append((name, raw.decode()))
    for name, identity in item['binary_identities'].items():
        raw = (Path(item['cache_directory']) / name).read_bytes()
        assert sha(raw) == identity['sha256'] and len(raw) == identity['bytes']
    inner = json.loads(snapshots['native-inspection.json'])
    native = json.loads(snapshots['owner-inspection.json'])
    assert set(native['imports']) == {'topal_tail_step_component_v1', 'topal_x86_tail_boundary_active_v1',
        'topal_x86_tail_boundary_call_v1', 'topal_x86_tail_checkpoint_call_v1',
        'topal_x86_tail_complete_call_v1', 'topal_x86_tail_boundary_fail_v1'}
    for field in ['owner_stack_paths', 'combined_paths']:
        paths = native[field]
        for name, cost in paths.items():
            if name.startswith('topal_x86_tail_'): continue
            assert cost == native['frames'][name] + 8 + max((paths[x] for x in native['calls'][name]), default=0)
    assert native['owner_stack_bytes'] == native['owner_stack_paths']['topal_tail_owner_invoke_v1']
    assert native['combined_normal_stack_bytes'] == native['combined_paths']['topal_tail_owner_invoke_v1'] <= 4096
    for name, key in [('topal_x86_tail_boundary_call_v1', 'maximum_source_stack_bytes'),
                      ('topal_x86_tail_checkpoint_call_v1', 'maximum_checkpoint_stack_bytes'),
                      ('topal_x86_tail_complete_call_v1', 'maximum_completion_stack_bytes')]:
        assert native['owner_stack_paths'][name] == 56
        assert native['combined_paths'][name] == 56 + inner[key]
    assert native['owner_stack_paths']['topal_x86_tail_boundary_fail_v1'] == 56
    assert native['combined_paths']['topal_x86_tail_boundary_fail_v1'] == 56
    line = snapshots['native.stdout.log'].decode()
    facts = {k:int(v) for k,v in re.findall(r'(\w+)=(\d+)',line)}
    assert facts == ({'admitted':501,'refused':2252,'fatal':0} if metadata['fixture']==0 else {'admitted':0,'refused':0,'fatal':1})
    group = owner_totals.setdefault(item['command'], {'clients':0,'admitted':0,'refused':0,'fatal':0,'stack':set(),'combined':set(),'fixtures':set()})
    group['clients'] += 1
    for key, value in facts.items(): group[key] += value
    group['stack'].add(native['owner_stack_bytes'])
    group['combined'].add(native['combined_normal_stack_bytes'])
    group['fixtures'].add(metadata['fixture'])
assert not owner_warnings, owner_warnings
for group in owner_totals.values():
    assert group['clients']==3 and group['fixtures']=={0,1,2}
print('OWNER_TOTALS', owner_totals)
print('OWNER_RECORDS', len(record['owner_components']), 'OWNER_TOOL_WARNINGS', owner_warnings)

kernel_totals = {}
normal_sequences = []
kernel_owner_objects = set()
for item in record['kernel_components']:
    snapshots = {name: read(entry) for name, entry in item['snapshots'].items()}
    owner = json.loads(snapshots['owner-component.json'])
    assert sha(snapshots['owner.ll']) == owner['llvm_sha256']
    assert sha(snapshots['runtime.ll']) == owner['runtime']['llvm_sha256']
    assert sha(snapshots['source.ll']) == owner['runtime']['source_step']['llvm_sha256']
    assert sha(snapshots['boundary.o']) == owner['runtime']['boundary_sha256']
    assert b'\x04S\x00\x08\x00\x01\x78\x10' in snapshots['boundary.o']
    assert b'\x04S\x00\x08\x00\x01\x78\x10' not in snapshots['kernel-owner.o']
    kernel_owner_objects.add(sha(snapshots['kernel-owner.o']))
    assert b'ResumeBootstrapRegionAfterLibrary {\n        require_true: true' in snapshots['root-model.txt']
    assert sha(snapshots['root.t']) == record['sources']['linux-kernel/labs/qemu/x86_64/tail-kernel-root.t']
    assert sha(snapshots['source.t']) == record['sources']['linux-kernel/kernel/algorithms/clear-prefix.t']
    for name, identity in item['binary_identities'].items():
        raw = (Path(item['cache_directory']) / name).read_bytes()
        assert sha(raw) == identity['sha256'] and len(raw) == identity['bytes']
    for name, raw in snapshots.items():
        if name.endswith('.command.json') and not name.endswith('qemu.command.json'):
            assert json.loads(raw)['exit_code'] == 0
        if name.endswith('.stderr.log') and raw:
            assert name.endswith('/gdb.stderr.log') and raw == b'warning: .debug_names not created by gdb; ignoring\n', (name, raw)
    inner = json.loads(snapshots['inner-inspection.json'])
    native = json.loads(snapshots['owner-inspection.json'])
    assert native['owner_stack_bytes'] == 240
    assert native['combined_normal_stack_bytes'] == 688
    assert native['combined_normal_stack_bytes'] + 8 + 48 + 80 == 824 <= 4096
    assert inner['maximum_source_stack_bytes'] == 488
    assert inner['maximum_checkpoint_stack_bytes'] == 312
    assert inner['maximum_completion_stack_bytes'] == 184
    modes = ['normal','source_abort','false_result','bad_key','busy','interrupts','bad_stack',
        'generation_exhaustion','frame_exhaustion','dirty_private']
    assert {name.split('/')[0] for name in snapshots if name.endswith('/guest-observations.json')} == set(modes)
    for mode in modes:
        observation = json.loads(snapshots[mode+'/guest-observations.json'])
        assert observation['mode'] == mode
        assert observation['retired_controller_bytes'] == 320
        assert b'TAIL_KERNEL_GDB_PASS' in snapshots[mode+'/gdb.stdout.log']
        assert b'TOPAL_KERNEL_BOOT' in snapshots[mode+'/serial.log']
        facts = observation['counts']
        assert facts['entry'] == 1
        assert facts['final'] + facts['fatal'] + facts['refused'] == 1
        assert facts['owner_resume'] == int(mode=='normal')
        assert facts['return'] == int(mode=='normal')
        assert facts['finalize'] == int(mode=='normal')
        assert facts['source'] == (65 if mode in ['normal','false_result'] else 1 if mode=='source_abort' else 0)
        assert facts['checkpoint'] == (65 if mode in ['normal','false_result'] else 0)
        assert facts['complete'] == int(mode in ['normal','false_result'])
        caller = observation['caller']
        assert caller['key'] == 4097
        for index, (start, size) in enumerate(caller['domains']):
            for other, length in caller['domains'][index+1:]:
                assert start+size <= other or other+length <= start
        walks = observation['walked_instructions']
        if mode == 'normal':
            assert walks['apply'] == 90 and walks['fatal'] == 0
            assert walks['return'] in [114,115]
            sequence = []
            for line in snapshots[mode+'/gdb.stdout.log'].decode().splitlines():
                match = re.search(r'(0x[0-9a-f]+) in', line)
                if 'region_library_apply_v1' not in line or not match: continue
                pc = int(match.group(1),16)
                if not sequence or pc != sequence[-1]: sequence.append(pc)
            assert len(sequence) == 205
            normal_sequences.append(sequence)
        else:
            assert walks == ({'apply':0,'return':0,'fatal':133} if mode=='source_abort' else {'apply':0,'return':0,'fatal':0})
        if mode in ['normal','false_result','source_abort']:
            assert 'topal_tail_owner_invoke_v1' in observation['traces'][0]['trace']
            assert 'topal_x86_systems_region_library_apply_v1' in observation['traces'][0]['trace']
        if mode in ['source_abort','false_result','generation_exhaustion','frame_exhaustion','dirty_private']:
            assert '_topal_kernel_entry' in observation['traces'][-1]['trace']
    kernel_totals[item['command']] = len(modes)
assert kernel_totals == {'tail-kernel':10, 'compiler':10}, kernel_totals
assert len(record['owner_components']) == 6
assert len(record['native_components']) == 124
assert len(kernel_owner_objects) == 1
assert len(normal_sequences) == 2 and normal_sequences[0] == normal_sequences[1]
print('KERNEL_VM_TOTALS', kernel_totals, 'CFI_ENTRY_RETURN_FATAL', [90,'114 instructions; 114/115 debugger stops',133])
print('KERNEL_READBACK_PASS')
