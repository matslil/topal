# Independent ELF readback: no compiler object/relocation inspection functions.
import struct

class Elf:
    def __init__(self, raw):
        self.raw = raw
        assert raw[:6] == b'\x7fELF\x02\x01'
        assert struct.unpack_from('<H',raw,18)[0] == 62
        shoff = struct.unpack_from('<Q',raw,40)[0]
        shstride, shnum, names = struct.unpack_from('<HHH',raw,58)
        assert shstride == 64
        self.sections = [dict(zip(['nameoff','type','flags','address','offset','size','link','info','align','stride'],
            struct.unpack_from('<IIQQQQIIQQ',raw,shoff+i*shstride))) for i in range(shnum)]
        text = self.data(names)
        for s in self.sections:
            s['name'] = self.string(text,s['nameoff'])
        self.tables = {}
        self.names = {}
        for index,s in enumerate(self.sections):
            if s['type'] != 2: continue
            assert s['stride'] == 24
            strings = self.data(s['link'])
            symbols = []
            for p in range(s['offset'],s['offset']+s['size'],24):
                n,info,other,section,value,size = struct.unpack_from('<IBBHQQ',raw,p)
                symbol = dict(name=self.string(strings,n),binding=info>>4,kind=info&15,
                    section=section,value=value,size=size,other=other)
                symbols.append(symbol)
                if section and section < len(self.sections):
                    self.names.setdefault(symbol['name'],[]).append(symbol)
            self.tables[index] = symbols
        phoff = struct.unpack_from('<Q',raw,32)[0]
        phstride, phnum = struct.unpack_from('<HH',raw,54)
        self.loads = []
        if phnum:
            assert phstride == 56
            for i in range(phnum):
                p = dict(zip(['type','flags','offset','address','physical','filesz','memsz','align'],
                    struct.unpack_from('<IIQQQQQQ',raw,phoff+i*phstride)))
                if p['type'] == 1: self.loads.append(p)
    @staticmethod
    def string(raw, offset):
        return raw[offset:raw.index(0,offset)].decode()
    def data(self, index):
        s = self.sections[index]
        return b'' if s['type']==8 else self.raw[s['offset']:s['offset']+s['size']]
    def one(self, name):
        values = self.names.get(name,[])
        assert len(values)==1,(name,len(values))
        return values[0]

def rebind(kernel_raw, binding, get_object):
    kernel = Elf(kernel_raw)
    occupied = []
    observed_markers = set()
    processing_flags = 0x200000 | 16 | 32
    assert struct.unpack_from('<Q',kernel_raw,24)[0] == kernel.one('_topal_kernel_entry')['value']
    placed = {p['marker']:p for p in binding['linked']}
    assert len(placed)==len(binding['linked'])
    assert [o['role'] for o in binding['objects']] == ['root','provider','source','runtime','owner','boundary','kernel-owner']
    for record in binding['objects']:
        raw = get_object(record['role'])
        assert sha(raw)==record['sha256']
        obj = Elf(raw)
        bases = {}
        sections = {}
        actual = {i for i,s in enumerate(obj.sections) if s['flags']&2 and s['size']}
        for r in record['sections']:
            marker = r['marker']
            a,b = kernel.one(marker+'.begin'),kernel.one(marker+'.end')
            oa,ob = obj.one(marker+'.begin'),obj.one(marker+'.end')
            assert oa['value']==0 and ob['value']==r['bytes'] and oa['section']==ob['section']
            assert a['binding']==b['binding']==0 and a['kind']==b['kind']==0
            start,end = a['value'],b['value']
            assert end-start == r['bytes'] and start%r['alignment']==0
            assert a['section']==b['section'] and a['size']==b['size']==0
            assert 0x200000<=start<end<=0x40000000 and (end<=0x400000 or start>=0x800000)
            assert all(end<=lo or start>=hi for lo,hi in occupied)
            occupied.append((start,end))
            original = obj.sections[oa['section']]
            linked = kernel.sections[a['section']]
            assert original['size']==r['bytes'] and original['flags']==r['flags']
            assert original['align']==r['alignment']
            assert (original['type']==8)==r['zero']==(linked['type']==8)
            assert linked['flags']&~processing_flags == r['flags']&~processing_flags
            assert linked['align']>=r['alignment']
            assert linked['address']<=start<end<=linked['address']+linked['size']
            assert oa['section'] not in bases
            bases[oa['section']]=start
            sections[oa['section']]=(r,a)
            observed_markers.update([marker+'.begin',marker+'.end'])
        assert set(bases)==actual
        for table in obj.tables.values():
            for sym in table:
                if sym['binding'] not in [1,2] or sym['section'] not in bases: continue
                actual_sym = kernel.one(sym['name'])
                assert actual_sym['value']==bases[sym['section']]+sym['value']
                assert actual_sym['size']==sym['size'] and actual_sym['kind']==sym['kind']
        for index,(r,a) in sections.items():
            data = bytearray(obj.data(index))
            for reloc in obj.sections:
                if reloc['type'] != 4 or reloc['info'] != index: continue
                assert reloc['stride']==24 and not r['zero']
                for p in range(reloc['offset'],reloc['offset']+reloc['size'],24):
                    offset,info,addend = struct.unpack_from('<QQq',obj.raw,p)
                    symbol = obj.tables[reloc['link']][info>>32]
                    if symbol['section'] in bases:
                        target = bases[symbol['section']]+symbol['value']
                    else:
                        assert symbol['section']==0 and symbol['binding']==1
                        target = kernel.one(symbol['name'])['value']
                    kind = info&0xffffffff
                    value = target+addend
                    if kind in [2,4]: value -= bases[index]+offset
                    assert kind in [1,2,4,10,11],kind
                    size = 8 if kind==1 else 4
                    encoded = value.to_bytes(size,'little',signed=kind in [2,4,11])
                    assert 0<=offset<=len(data)-size
                    data[offset:offset+size]=encoded
            start,end = a['value'],a['value']+r['bytes']
            linked = kernel.sections[a['section']]
            if r['zero']:
                assert not data
            else:
                offset = start-linked['address']
                assert kernel.data(a['section'])[offset:offset+len(data)]==data
            loads = [p for p in kernel.loads if p['address']<=start<end<=p['address']+p['memsz']]
            assert len(loads)==1
            load = loads[0]
            assert load['flags']==4 | (2 if r['flags']&1 else 0) | (1 if r['flags']&4 else 0)
            loaded = kernel.raw[load['offset']:load['offset']+load['filesz']]
            offset = start-load['address']
            if r['zero']: assert not any(loaded[offset:offset+r['bytes']])
            else: assert loaded[offset:offset+len(data)]==data
            assert placed[r['marker']]==dict(marker=r['marker'],address=start,bytes=r['bytes'],
                alignment=r['alignment'],flags=r['flags'],sha256=sha(bytes(data)))
    assert len(placed)==len(occupied)
    assert {n for n in kernel.names if n.startswith('topal.tail-inputs.')}==observed_markers
    return len(occupied)
