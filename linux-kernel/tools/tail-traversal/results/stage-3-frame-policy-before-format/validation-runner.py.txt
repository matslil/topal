def save_bytes(directory, name, raw):
    compressed = gzip.compress(raw, mtime=0)
    destination = directory / (name + '.gz')
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(compressed)
    assert gzip.decompress(compressed) == raw
    return {'file': str(destination.relative_to(out)), 'sha256': sha(compressed), 'uncompressed_sha256': sha(raw)}

seen = set()
def collect_native(command_name):
    for metadata_path in sorted(scratch.glob('tail-step-*/component.json')):
        if str(metadata_path) in seen:
            continue
        seen.add(str(metadata_path))
        directory = metadata_path.parent
        metadata = json.loads(metadata_path.read_text())
        assert metadata['llvm_sha256'] == sha((directory / 'step.ll').read_bytes())
        item = {'command': command_name, 'cache_directory': str(directory), 'metadata': metadata,
                'snapshots': {}, 'binary_identities': {}}
        for filename in ['step.ll', 'tail_step_bindings.h', 'component.json', 'runtime.ll', 'runtime.json', 'native-inspection.json'] + [label + suffix for label in ['llc', 'runtime-llc', 'source-disassembly', 'runtime-disassembly', 'cc', 'native', 'gdb'] for suffix in ['.stdout.log', '.stderr.log', '.command.json']]:
            path = directory / filename
            if path.exists():
                item['snapshots'][filename] = save_bytes(out / 'native-components' / directory.name, filename, path.read_bytes())
        for filename in ['step.o', 'client', 'boundary.o', 'runtime.o']:
            path = directory / filename
            if path.exists():
                raw = path.read_bytes()
                item['binary_identities'][filename] = {'sha256': sha(raw), 'bytes': len(raw), 'retention': 'external disk cache; exact LLVM/C/header/command inputs archived'}
        record['native_components'].append(item)
