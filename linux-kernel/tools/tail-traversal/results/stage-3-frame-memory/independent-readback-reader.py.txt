def topology(o,key):
    # Different oracle from the in-guest interval sweep: admit complete frames
    # from RAM, then reject every boundary crossing or reservation overlap.
    candidates=set();boundaries=set()
    for start,length,kind in o['e820']:
        boundaries.update([start,start+length])
        if kind==1:candidates.update(range((start+4095)//4096,(start+length)//4096))
    expected=[]
    for f in sorted(candidates):
        a,b=f*4096,(f+1)*4096
        if any(a<x<b for x in boundaries):continue
        if any(kind!=1 and a<start+length and start<b for start,length,kind in o['e820']):continue
        if any(a<end and start<b for start,end in o['reservations']):continue
        expected.append(f)
    assert expected==o[key] and len(expected)==o['n']
